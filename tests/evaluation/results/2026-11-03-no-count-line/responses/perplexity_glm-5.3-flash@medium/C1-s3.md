## Headline

Replace all 11 `print` calls with calls to a module logger, and configure it once in `main()` with `basicConfig(level=logging.INFO)`. Supporting points:

1. Records written at WARNING and ERROR stay visible, because INFO is the minimum level, not the only level [1].
2. Output goes to stderr by default, so the current stream behavior does not change [1].
3. Error records gain tracebacks and timestamps without any changes to the call sites.

## Background

The service `service.py` (129 lines) logs startup, database, request, and error events through `print` to stdout and stderr. The request is to move to the standard `logging` module at INFO level. This removes unstructured output and adds levels, timestamps, and names to each record.

## Relevance

Unstructured `print` output cannot be filtered or routed by severity. The change is small, and it unlocks level based filtering for the slow request and error paths the service already tracks.

## Information

**Evidence**

Per the call sites, the 11 records map to three levels:

- INFO, 8 records: "opening database", "checking schema", "schema ok", "order not found", "listed N orders", "order N is now S", "request", and the four startup or shutdown records [no source, read from the file].
- WARNING, 3 records: bad status value, slow request over 500 ms, and bad client JSON [no source, read from the file].
- ERROR, 1 record: the exception handler in `do_GET` [no source, read from the file].

The module sends records at or above the configured level to stderr by default [1]. Therefore the two `print(..., file=sys.stderr)` sites need no stream change. The `sys` import becomes unused and is removed.

Design choices: one module logger `LOG = logging.getLogger(__name__)`, one `basicConfig` call in `main()`, lazy `%s` placeholders instead of f-strings, and `LOG.exception` in `do_GET` so the traceback is captured [1].

```python
"""Order status service: a small HTTP API over the orders database."""

import json
import logging
import os
import sqlite3
import time
from http.server import BaseHTTPRequestHandler, HTTPServer

LOG = logging.getLogger(__name__)

DB_PATH = os.environ.get("ORDERS_DB", "orders.db")
PORT = int(os.environ.get("PORT", "8080"))
SLOW_MS = 500


def connect():
    LOG.info("opening database %s", DB_PATH)
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db(conn):
    LOG.info("checking schema")
    conn.execute(
        "CREATE TABLE IF NOT EXISTS orders ("
        "id INTEGER PRIMARY KEY, customer TEXT, status TEXT, total REAL, updated REAL)"
    )
    conn.commit()
    LOG.info("schema ok")


def get_order(conn, order_id):
    row = conn.execute("SELECT * FROM orders WHERE id = ?", (order_id,)).fetchone()
    if row is None:
        LOG.info("order not found: %s", order_id)
        return None
    return dict(row)


def list_orders(conn, customer):
    rows = conn.execute(
        "SELECT * FROM orders WHERE customer = ? ORDER BY updated DESC", (customer,)
    ).fetchall()
    LOG.info("listed %d orders for %s", len(rows), customer)
    return [dict(r) for r in rows]


def update_status(conn, order_id, status):
    allowed = {"new", "paid", "shipped", "cancelled"}
    if status not in allowed:
        LOG.warning("bad status %s for order %s", status, order_id)
        return False
    conn.execute(
        "UPDATE orders SET status = ?, updated = ? WHERE id = ?",
        (status, time.time(), order_id),
    )
    conn.commit()
    LOG.info("order %d is now %s", order_id, status)
    return True


class Handler(BaseHTTPRequestHandler):
    conn = None

    def send_json(self, code, body):
        data = json.dumps(body).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        started = time.monotonic()
        parts = self.path.strip("/").split("/")
        try:
            if len(parts) == 2 and parts[0] == "orders":
                order = get_order(self.conn, int(parts[1]))
                if order is None:
                    self.send_json(404, {"error": "not found"})
                else:
                    self.send_json(200, order)
            elif len(parts) == 2 and parts[0] == "customers":
                self.send_json(200, list_orders(self.conn, parts[1]))
            else:
                self.send_json(404, {"error": "no route"})
        except Exception:
            LOG.exception("error in GET %s", self.path)
            self.send_json(500, {"error": "internal"})
        elapsed = (time.monotonic() - started) * 1000
        if elapsed > SLOW_MS:
            LOG.warning("slow request: %s %d ms", self.path, round(elapsed))

    def do_POST(self):
        parts = self.path.strip("/").split("/")
        length = int(self.headers.get("Content-Length", "0"))
        try:
            body = json.loads(self.rfile.read(length) or b"{}")
        except json.JSONDecodeError:
            LOG.warning("bad JSON from %s", self.client_address[0])
            self.send_json(400, {"error": "bad json"})
            return
        if len(parts) == 3 and parts[0] == "orders" and parts[2] == "status":
            ok = update_status(self.conn, int(parts[1]), body.get("status", ""))
            self.send_json(200 if ok else 400, {"ok": ok})
        else:
            self.send_json(404, {"error": "no route"})

    def log_message(self, format, *args):
        LOG.info("request: %s %s", self.address_string(), format % args)


def main():
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    LOG.info("starting order service on port %d", PORT)
    conn = connect()
    init_db(conn)
    Handler.conn = conn
    server = HTTPServer(("", PORT), Handler)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        LOG.info("stopping")
    finally:
        conn.close()
        LOG.info("database closed")


if __name__ == "__main__":
    main()
```
