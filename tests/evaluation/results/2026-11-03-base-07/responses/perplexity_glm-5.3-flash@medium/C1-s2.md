# Headline

Swap all 14 print calls in service.py for one module-level logger, and configure logging at level INFO.

- Eleven calls map to INFO, because their original text encodes no severity.
- The bad status and slow request messages map to WARNING. The caught GET failure maps to ERROR with a traceback.
- The setup uses logging.basicConfig with a one-line format, so the log stream and level live in one place.
- The import of sys becomes unused after the swap and is removed.

# Background

The attached file service.py is a 129-line HTTP service over SQLite [attached file, service.py]. It reports events through 14 print calls: 12 on stdout and 2 on stderr (the GET error uses file=sys.stderr). The request asks to replace them with the logging module at INFO level.

# Relevance

Print output carries no level, timestamp, or logger name, so log tools cannot filter or route it. A standard logger adds those fields at a small code cost, and it keeps every call site unchanged in behavior.

Notes on three choices:

1. The slow request message becomes WARNING [judgment]. It flags an anomaly above a 500 ms threshold, and a threshold breach is not normal operation.
2. The literal "WARNING:" and "ERROR:" prefixes disappear. The level field carries that meaning instead, and the level field is what filters read [inference].
3. LOG.exception records the traceback for the caught GET failure, which replaces the bare error text [1].

The setup block is:

```python
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s %(levelname)s %(name)s %(message)s",
)
LOG = logging.getLogger(__name__)
```

basicConfig adds a StreamHandler to stderr by default [1]. Therefore the two stderr prints move to the same stream as the rest, and the import of sys goes away. The logger name will be "__main__" when the file runs directly, because __name__ equals "__main__" at the entry point [inference]. Each call uses lazy %s formatting instead of an f-string, so the string builds only when the record passes the level filter [1].

```python
"""Order status service: a small HTTP API over the orders database."""

import json
import logging
import os
import sqlite3
import time
from http.server import BaseHTTPRequestHandler, HTTPServer

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s %(levelname)s %(name)s %(message)s",
)
LOG = logging.getLogger(__name__)

DB_PATH = os.environ.get("ORDERS_DB", "orders.db")
PORT = int(os.environ.get("PORT", "8080"))
SLOW_MS = 500


def connect():
    LOG.info("opening database %s", DB_PATH)
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db(conn):
    LOG.info("checking schema")
    conn.execute(
        "CREATE TABLE IF NOT EXISTS orders ("
        "id INTEGER PRIMARY KEY, customer TEXT, status TEXT, total REAL, updated REAL)"
    )
    conn.commit()
    LOG.info("schema ok")


def get_order(conn, order_id):
    row = conn.execute("SELECT * FROM orders WHERE id = ?", (order_id,)).fetchone()
    if row is None:
        LOG.info("order not found: %s", order_id)
        return None
    return dict(row)


def list_orders(conn, customer):
    rows = conn.execute(
        "SELECT * FROM orders WHERE customer = ? ORDER BY updated DESC", (customer,)
    ).fetchall()
    LOG.info("listed %d orders for %s", len(rows), customer)
    return [dict(r) for r in rows]


def update_status(conn, order_id, status):
    allowed = {"new", "paid", "shipped", "cancelled"}
    if status not in allowed:
        LOG.warning("bad status %r for order %s", status, order_id)
        return False
    conn.execute(
        "UPDATE orders SET status = ?, updated = ? WHERE id = ?",
        (status, time.time(), order_id),
    )
    conn.commit()
    LOG.info("order %s is now %s", order_id, status)
    return True


class Handler(BaseHTTPRequestHandler):
    conn = None

    def send_json(self, code, body):
        data = json.dumps(body).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        started = time.monotonic()
        parts = self.path.strip("/").split("/")
        try:
            if len(parts) == 2 and parts[0] == "orders":
                order = get_order(self.conn, int(parts[1]))
                if order is None:
                    self.send_json(404, {"error": "not found"})
                else:
                    self.send_json(200, order)
            elif len(parts) == 2 and parts[0] == "customers":
                self.send_json(200, list_orders(self.conn, parts[1]))
            else:
                self.send_json(404, {"error": "no route"})
        except Exception as error:
            LOG.exception("error in GET %s: %s", self.path, error)
            self.send_json(500, {"error": "internal"})
        elapsed = (time.monotonic() - started) * 1000
        if elapsed > SLOW_MS:
            LOG.warning("slow request: %s %d ms", self.path, round(elapsed))

    def do_POST(self):
        parts = self.path.strip("/").split("/")
        length = int(self.headers.get("Content-Length", "0"))
        try:
            body = json.loads(self.rfile.read(length) or b"{}")
        except json.JSONDecodeError:
            LOG.info("bad JSON from %s", self.client_address[0])
            self.send_json(400, {"error": "bad json"})
            return
        if len(parts) == 3 and parts[0] == "orders" and parts[2] == "status":
            ok = update_status(self.conn, int(parts[1]), body.get("status", ""))
            self.send_json(200 if ok else 400, {"ok": ok})
        else:
            self.send_json(404, {"error": "no route"})

    def log_message(self, format, *args):
        LOG.info("request: %s %s", self.address_string(), format % args)


def main():
    LOG.info("starting order service on port %d", PORT)
    conn = connect()
    init_db(conn)
    Handler.conn = conn
    server = HTTPServer(("", PORT), Handler)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        LOG.info("stopping")
    finally:
        conn.close()
        LOG.info("database closed")


if __name__ == "__main__":
    main()
```
