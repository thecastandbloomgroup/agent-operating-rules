## Headline

The rewrite below replaces all 14 print statements in service.py with the standard logging module, set once at INFO level [1].

One basicConfig call in main() sets the INFO threshold, the message format, and the stderr destination. Twelve untagged prints become logger.info, and the two tagged prints become logger.warning and logger.exception. Four executable tests prove the new behavior.

## Background

service.py is a 129-line HTTP order API with 14 print statements across startup, database, and request code [1]. The request: replace them with the standard logging module at INFO level. Two prints carry severity tags in their text, one WARNING and one ERROR.

## Relevance

Print output has no level control, no timestamps, and no routing. Logging supplies all three from the standard library, so the change adds no dependency [2].

## Information

Severity mapping, with levels deduced from the tags in the source [1]:

| Original print | Replacement | Level |
|---|---|---|
| 12 untagged messages (startup, database, request) | logger.info with %s args | INFO |
| "WARNING: bad status" in update_status | logger.warning | WARNING |
| "ERROR in GET" in the except block | logger.exception | ERROR |

Design points:

- Arguments use lazy %s formatting, so the string is built only when the record is emitted [2].
- logger.exception emits at ERROR level and attaches the traceback (the error call stack) [2].
- The default basicConfig handler writes to stderr, which preserves the old ERROR destination [2].
- The diff touches only log lines and the removed sys import; routes and schema are unchanged [1].
- assertLogs, the unittest helper that captures records, enables the tests without handler setup [3].

```python
"""Order status service: a small HTTP API over the orders database."""

import json
import logging
import os
import sqlite3
import time
from http.server import BaseHTTPRequestHandler, HTTPServer

logger = logging.getLogger(__name__)

DB_PATH = os.environ.get("ORDERS_DB", "orders.db")
PORT = int(os.environ.get("PORT", "8080"))
SLOW_MS = 500


def connect():
    logger.info("opening database %s", DB_PATH)
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db(conn):
    logger.info("checking schema")
    conn.execute(
        "CREATE TABLE IF NOT EXISTS orders ("
        "id INTEGER PRIMARY KEY, customer TEXT, status TEXT, total REAL, updated REAL)"
    )
    conn.commit()
    logger.info("schema ok")


def get_order(conn, order_id):
    row = conn.execute("SELECT * FROM orders WHERE id = ?", (order_id,)).fetchone()
    if row is None:
        logger.info("order not found: %s", order_id)
        return None
    return dict(row)


def list_orders(conn, customer):
    rows = conn.execute(
        "SELECT * FROM orders WHERE customer = ? ORDER BY updated DESC", (customer,)
    ).fetchall()
    logger.info("listed %d orders for %s", len(rows), customer)
    return [dict(r) for r in rows]


def update_status(conn, order_id, status):
    allowed = {"new", "paid", "shipped", "cancelled"}
    if status not in allowed:
        logger.warning("bad status %s for order %s", status, order_id)
        return False
    conn.execute(
        "UPDATE orders SET status = ?, updated = ? WHERE id = ?",
        (status, time.time(), order_id),
    )
    conn.commit()
    logger.info("order %s is now %s", order_id, status)
    return True


class Handler(BaseHTTPRequestHandler):
    conn = None

    def send_json(self, code, body):
        data = json.dumps(body).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        started = time.monotonic()
        parts = self.path.strip("/").split("/")
        try:
            if len(parts) == 2 and parts[0] == "orders":
                order = get_order(self.conn, int(parts[1]))
                if order is None:
                    self.send_json(404, {"error": "not found"})
                else:
                    self.send_json(200, order)
            elif len(parts) == 2 and parts[0] == "customers":
                self.send_json(200, list_orders(self.conn, parts[1]))
            else:
                self.send_json(404, {"error": "no route"})
        except Exception:
            logger.exception("error in GET %s", self.path)
            self.send_json(500, {"error": "internal"})
        elapsed = (time.monotonic() - started) * 1000
        if elapsed > SLOW_MS:
            logger.info("slow request: %s %d ms", self.path, round(elapsed))

    def do_POST(self):
        parts = self.path.strip("/").split("/")
        length = int(self.headers.get("Content-Length", "0"))
        try:
            body = json.loads(self.rfile.read(length) or b"{}")
        except json.JSONDecodeError:
            logger.info("bad JSON from %s", self.client_address[0])
            self.send_json(400, {"error": "bad json"})
            return
        if len(parts) == 3 and parts[0] == "orders" and parts[2] == "status":
            ok = update_status(self.conn, int(parts[1]), body.get("status", ""))
            self.send_json(200 if ok else 400, {"ok": ok})
        else:
            self.send_json(404, {"error": "no route"})

    def log_message(self, format, *args):
        logger.info("request: %s %s", self.address_string(), format % args)


def main():
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s %(levelname)s %(name)s %(message)s",
    )
    logger.info("starting order service on port %s", PORT)
    conn = connect()
    init_db(conn)
    Handler.conn = conn
    server = HTTPServer(("", PORT), Handler)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        logger.info("stopping")
    finally:
        conn.close()
        logger.info("database closed")


if __name__ == "__main__":
    main()
```

```python
"""Tests for the logging behavior of service.py."""

import sqlite3
import unittest

import service


def make_conn():
    conn = sqlite3.connect(":memory:")
    conn.row_factory = sqlite3.Row
    service.init_db(conn)
    conn.execute(
        "INSERT INTO orders (id, customer, status, total, updated) "
        "VALUES (1, 'ada', 'new', 9.5, 0.0)"
    )
    conn.commit()
    return conn


class FakeHandler(service.Handler):
    def __init__(self, path):
        self.path = path
        self.conn = make_conn()
        self.sent = []

    def send_json(self, code, body):
        self.sent.append((code, body))


class LoggingTest(unittest.TestCase):
    def test_list_orders_logs_info(self):
        with self.assertLogs(service.logger, level="INFO") as caught:
            service.list_orders(make_conn(), "ada")
        self.assertIn("INFO", caught.output[0])
        self.assertIn("listed 1 orders for ada", caught.output[0])

    def test_bad_status_logs_warning(self):
        with self.assertLogs(service.logger, level="WARNING") as caught:
            ok = service.update_status(make_conn(), 1, "bogus")
        self.assertFalse(ok)
        self.assertIn("WARNING", caught.output[0])

    def test_missing_order_logs_info(self):
        with self.assertLogs(service.logger, level="INFO") as caught:
            result = service.get_order(make_conn(), 99)
        self.assertIsNone(result)
        self.assertIn("order not found: 99", caught.output[0])

    def test_get_error_logs_error_and_sends_500(self):
        handler = FakeHandler("/orders/1")
        handler.conn.close()
        with self.assertLogs(service.logger, level="ERROR") as caught:
            handler.do_GET()
        self.assertIn("ERROR", caught.output[0])
        self.assertEqual(500, handler.sent[0][0])


if __name__ == "__main__":
    unittest.main()
```
