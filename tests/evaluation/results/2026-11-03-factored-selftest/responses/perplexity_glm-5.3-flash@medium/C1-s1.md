```python
"""Order status service: a small HTTP API over the orders database."""

import json
import logging
import os
import sqlite3
import time
from http.server import BaseHTTPRequestHandler, HTTPServer

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s %(levelname)s %(name)s %(message)s",
)
log = logging.getLogger("orders")

DB_PATH = os.environ.get("ORDERS_DB", "orders.db")
PORT = int(os.environ.get("PORT", "8080"))
SLOW_MS = 500


def connect():
    log.info("opening database %s", DB_PATH)
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db(conn):
    log.info("checking schema")
    conn.execute(
        "CREATE TABLE IF NOT EXISTS orders ("
        "id INTEGER PRIMARY KEY, customer TEXT, status TEXT, total REAL, updated REAL)"
    )
    conn.commit()
    log.info("schema ok")


def get_order(conn, order_id):
    row = conn.execute("SELECT * FROM orders WHERE id = ?", (order_id,)).fetchone()
    if row is None:
        log.warning("order not found: %s", order_id)
        return None
    return dict(row)


def list_orders(conn, customer):
    rows = conn.execute(
        "SELECT * FROM orders WHERE customer = ? ORDER BY updated DESC", (customer,)
    ).fetchall()
    log.info("listed %d orders for %s", len(rows), customer)
    return [dict(r) for r in rows]


def update_status(conn, order_id, status):
    allowed = {"new", "paid", "shipped", "cancelled"}
    if status not in allowed:
        log.warning("bad status %s for order %d", status, order_id)
        return False
    conn.execute(
        "UPDATE orders SET status = ?, updated = ? WHERE id = ?",
        (status, time.time(), order_id),
    )
    conn.commit()
    log.info("order %d is now %s", order_id, status)
    return True


class Handler(BaseHTTPRequestHandler):
    conn = None

    def send_json(self, code, body):
        data = json.dumps(body).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        started = time.monotonic()
        parts = self.path.strip("/").split("/")
        try:
            if len(parts) == 2 and parts[0] == "orders":
                order = get_order(self.conn, int(parts[1]))
                if order is None:
                    self.send_json(404, {"error": "not found"})
                else:
                    self.send_json(200, order)
            elif len(parts) == 2 and parts[0] == "customers":
                self.send_json(200, list_orders(self.conn, parts[1]))
            else:
                self.send_json(404, {"error": "no route"})
        except Exception:
            log.exception("error in GET %s", self.path)
            self.send_json(500, {"error": "internal"})
        elapsed = (time.monotonic() - started) * 1000
        if elapsed > SLOW_MS:
            log.info("slow request: %s %d ms", self.path, round(elapsed))

    def do_POST(self):
        parts = self.path.strip("/").split("/")
        length = int(self.headers.get("Content-Length", "0"))
        try:
            body = json.loads(self.rfile.read(length) or b"{}")
        except json.JSONDecodeError:
            log.warning("bad JSON from %s", self.client_address[0])
            self.send_json(400, {"error": "bad json"})
            return
        if len(parts) == 3 and parts[0] == "orders" and parts[2] == "status":
            ok = update_status(self.conn, int(parts[1]), body.get("status", ""))
            self.send_json(200 if ok else 400, {"ok": ok})
        else:
            self.send_json(404, {"error": "no route"})

    def log_message(self, format, *args):
        log.info("request: %s %s", self.address_string(), format % args)


def main():
    log.info("starting order service on port %d", PORT)
    conn = connect()
    init_db(conn)
    Handler.conn = conn
    server = HTTPServer(("", PORT), Handler)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        log.info("stopping")
    finally:
        conn.close()
        log.info("database closed")


if __name__ == "__main__":
    main()
```

# test_logging.py  (run: python test_logging.py)
import io, logging, re, sys, tempfile, unittest
from unittest import mock
import service

LINE = re.compile(r"^\d{4}-\d{2}-\d{2} .+ (INFO|WARNING|ERROR) orders ")


class TestAllOutputIsLogged(unittest.TestCase):
    def capture(self, ops):
        out, err = io.StringIO(), io.StringIO()
        with mock.patch.object(sys, "stdout", out), \
             mock.patch.object(sys, "stderr", err), \
             mock.patch.object(service, "DB_PATH", "unused.db"), \
             tempfile.TemporaryDirectory() as tmp:
            logging.basicConfig(stream=err, level=logging.INFO, force=True)
            with mock.patch.object(service, "DB_PATH", tmp + "/orders.db"):
                ops()
        return out.getvalue(), err.getvalue()

    def test_operations(self):
        def ops():
            conn = service.connect()
            service.init_db(conn)
            self.assertIsNone(service.get_order(conn, 7))
            service.list_orders(conn, "alice")
            self.assertTrue(service.update_status(conn, 7, "new"))
            self.assertFalse(service.update_status(conn, 7, "junk"))
            conn.close()
        out, err = self.capture(ops)
        self.assertEqual(out, "")
        lines = err.getvalue().splitlines()
        self.assertEqual(len(lines), 7)
        for line in lines:
            self.assertRegex(line, LINE)
        self.assertEqual(err.getvalue().count("INFO"), 5)
        self.assertEqual(err.getvalue().count("WARNING"), 2)
